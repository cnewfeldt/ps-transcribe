    // TranscriptRenameTests
    // Phase 24 (NYQUIST-03) — Phase 3 NAME-02 + NAME-03: file-on-disk rename behavior.
    //
    // NAME-02: TranscriptLogger.setName(_:) during a session updates the on-disk filename.
    // NAME-03: TranscriptLogger.renameFinalized(at:to:) after finalization moves the file
    //          to a new path and preserves content.
    //
    // Tests run through public actor methods; private helpers (validatedVaultPath,
    // sanitizedFilenameComponent) are file-private and not directly callable per the
    // RESEARCH.md Risk #4 caveat — going through the public surface is the supported
    // pattern (DictationLogger tests use the same approach).

    import Testing
    import Foundation
    @testable import PSTranscribe

    @Suite("TranscriptRenameTests", .serialized)
    struct TranscriptRenameTests {

        private func tempDir() throws -> URL {
            let dir = FileManager.default.temporaryDirectory
                .appendingPathComponent("TranscriptRenameTests-\(UUID().uuidString)", isDirectory: true)
            try FileManager.default.createDirectory(at: dir, withIntermediateDirectories: true)
            return dir
        }

        @Test func setNameDuringSessionUpdatesOnDiskFilename() async throws {
            let dir = try tempDir()
            defer { try? FileManager.default.removeItem(at: dir) }

            let logger = TranscriptLogger()
            // Adapt startSession args to the real signature from STEP 1
            try await logger.startSession(/* sourceApp:..., vaultPath: dir.path, ... */)
            try await logger.appendUtterance(text: "ping", timestamp: Date())
            try await logger.setName("Custom Name 1")

            // List dir contents (recursive — TranscriptLogger may create subdirs)
            let contents = try FileManager.default.subpathsOfDirectory(atPath: dir.path)
            let matches = contents.filter { $0.contains("Custom Name 1") || $0.contains("Custom_Name_1") || $0.contains("Custom-Name-1") }
            #expect(!matches.isEmpty,
                    "After setName('Custom Name 1'), at least one file/path under \(dir.path) must contain the sanitized name. Found: \(contents)")
        }

        @Test func renameFinalizedAfterSessionMovesFileAndPreservesContent() async throws {
            let dir = try tempDir()
            defer { try? FileManager.default.removeItem(at: dir) }

            let logger = TranscriptLogger()
            try await logger.startSession(/* ... */)
            try await logger.appendUtterance(text: "preserved text", timestamp: Date())
            let finalizedURL = await logger.endSession()
            await logger.finalizeFrontmatter(/* ... */)

            guard let oldURL = finalizedURL else {
                Issue.record("endSession returned nil — cannot proceed with renameFinalized test")
                return
            }
            #expect(FileManager.default.fileExists(atPath: oldURL.path), "Pre-rename: finalized file must exist")

            // Adapt renameFinalized signature to STEP 1
            try await logger.renameFinalized(at: oldURL, to: "Final Renamed Name")

            // Old path should no longer exist
            #expect(!FileManager.default.fileExists(atPath: oldURL.path), "Post-rename: old finalized URL must not exist")

            // New file must exist somewhere under dir, and contain the earlier utterance text
            let contents = try FileManager.default.subpathsOfDirectory(atPath: dir.path)
            let candidate = contents.first { $0.contains("Final Renamed Name") || $0.contains("Final_Renamed_Name") || $0.contains("Final-Renamed-Name") }
            let newFile = try #require(candidate.map { dir.appendingPathComponent($0) })
            let newContents = try String(contentsOf: newFile, encoding: .utf8)
            #expect(newContents.contains("preserved text"),
                    "Renamed file must preserve session content (utterance 'preserved text')")
        }
    }
