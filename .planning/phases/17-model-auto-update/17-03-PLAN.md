```swift
extension TranscriptionEngine {
    /// Hot-swap ASR + VAD models after the on-disk directory has been replaced atomically.
    /// MUST be called only when isRunning == false. Caller (ModelUpdateService) is responsible
    /// for the guard via SessionCoordinator.anySessionActive.
    /// Mirrors prepareModels() at lines 79-103, with an explicit nil-out step so ARC drops
    /// the previous MLModel handles before the new instances load. Per RESEARCH Pitfall #4
    /// the AsrModels.downloadAndLoad call is a no-op for the network when files already exist
    /// on disk (DownloadUtils.swift:192-204 allModelsExist short-circuit).
    func reloadModels() async throws {
        asrManager = nil
        vadManager = nil
        modelsReady = false

        let models = try await AsrModels.downloadAndLoad(version: .v3)
        let asr = AsrManager(config: .default)
        try await asr.loadModels(models)
        self.asrManager = asr

        let vad = try await VadManager()
        self.vadManager = vad

        modelsReady = true
        assetStatus = "Ready"
    }
}
```

```swift
// Continues from .verifying state set at end of Plan 17-02's runDownload.
// Splits the prior monolithic downloadAndApply into:
//   downloadAndApply() → preflight + runDownload → applySwap()
// applySwap is the new step.

private func applySwap(_ manifest: ModelManifest) async {
    // 1. Apply-deferral gate (D-19 / MODEL-07)
    if sessionCoordinator?.anySessionActive == true {
        // Stash the manifest for resumption; poll until session ends.
        await waitForSessionEnd()
    }

    isApplying = true
    defer { isApplying = false }
    updateState = .applying

    do {
        // 2. Rotate previous failed-staging dir (D-15 — keep one cycle of forensics)
        rotatePriorFailedStaging()

        // 3. Atomic swap via FileManager.replaceItem
        let timestamp = ISO8601DateFormatter().string(from: Date())
            .replacingOccurrences(of: ":", with: "")
        let backupName = "parakeet-tdt-0.6b-v3-backup-\(timestamp)"
        var resultingURL: NSURL?
        try FileManager.default.replaceItem(
            at: modelDirectory,
            withItemAt: stagingDirectory,
            backupItemName: backupName,
            options: [.usingNewMetadataOnly],
            resultingItemURL: &resultingURL
        )

        // 4. Hot-swap models (RESEARCH Pattern reloadModels())
        do {
            try await transcriptionEngine?.reloadModels()
        } catch {
            // Reload failed — rollback by reversing the swap with replaceItem (or moving the backup back)
            try rollbackSwap(backupName: backupName)
            updateState = .failed(message: "Reload failed: \(error.localizedDescription)")
            return
        }

        // 5. Cleanup the backup directory (best-effort)
        let backupURL = modelsRoot.appendingPathComponent(backupName)
        try? FileManager.default.removeItem(at: backupURL)

        // 6. Persist new version (MODEL-05)
        settings?.installedModelVersion = manifest.version

        // 7. Update state
        updateState = .applied(version: manifest.version)
    } catch {
        // Swap itself failed — staging is intact, production untouched, no rollback needed.
        // Move staging to a forensic-failed dir per D-15 if useful.
        moveStagingToFailed(error: error)
        updateState = .failed(message: "Swap failed: \(error.localizedDescription)")
    }
}

private func waitForSessionEnd() async {
    while sessionCoordinator?.anySessionActive == true {
        try? await Task.sleep(for: .milliseconds(500))
    }
}

private func rotatePriorFailedStaging() {
    // Find any existing parakeet-tdt-0.6b-v3-failed-* directory and remove it.
    guard let entries = try? FileManager.default.contentsOfDirectory(at: modelsRoot,
                                                                     includingPropertiesForKeys: nil) else { return }
    for entry in entries where entry.lastPathComponent.hasPrefix("parakeet-tdt-0.6b-v3-failed-") {
        try? FileManager.default.removeItem(at: entry)
    }
}

private func moveStagingToFailed(error: Error) {
    let timestamp = ISO8601DateFormatter().string(from: Date())
        .replacingOccurrences(of: ":", with: "")
    let failedDir = modelsRoot.appendingPathComponent("parakeet-tdt-0.6b-v3-failed-\(timestamp)")
    if FileManager.default.fileExists(atPath: stagingDirectory.path) {
        try? FileManager.default.moveItem(at: stagingDirectory, to: failedDir)
    }
}

private func rollbackSwap(backupName: String) throws {
    // After replaceItem, production now contains the new (broken) model and
    // the backup contains the prior model. Reverse: move broken aside, move backup back.
    let backupURL = modelsRoot.appendingPathComponent(backupName)
    let timestamp = ISO8601DateFormatter().string(from: Date())
        .replacingOccurrences(of: ":", with: "")
    let brokenDir = modelsRoot.appendingPathComponent("parakeet-tdt-0.6b-v3-failed-\(timestamp)")
    try FileManager.default.moveItem(at: modelDirectory, to: brokenDir)
    try FileManager.default.moveItem(at: backupURL, to: modelDirectory)
}
```

| filesystem (staging → production) | Atomic swap on APFS replaces production model with verified staging |
